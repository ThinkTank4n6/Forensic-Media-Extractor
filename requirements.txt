# Requirements for the forensic recovery tool (fat32_core, media_core,
# media_gui, db_extract, c4all_export, projectvic_export).
#
#   python3 -m venv .venv
#   .venv/bin/pip install -r requirements.txt        # Linux / macOS
#   .venv\Scripts\pip install -r requirements.txt    # Windows
#
# Python 3.9 or newer. The syntax would parse on 3.8, but the carve and the
# report stop worker pools with shutdown(cancel_futures=True), which 3.9
# added; on 3.8 a canceled run fails instead of stopping. pyproject.toml
# says the same.
#
# Only Flask is required to start. Everything else adds capability, and
# this file installs all of it, because an examiner's machine should have
# all of it: a missing optional package does not fail a run, it quietly
# narrows what the run can do -- no HEIC, no 7z, no encrypted E01. Each
# block says what you lose by leaving it out. For Flask alone, use
# requirements-minimal.txt. `python -m fme doctor` lists what is installed.
#
# Companion files:
#   requirements-minimal.txt  Flask only - use when pytsk3 or libewf will not
#                             build and you need the tool working now
#   requirements-vm.txt       VMDK / VHD / QCOW support
#   requirements-dev.txt      everything above plus the test dependencies

# ---------------------------------------------------------------------------
# Required for the web GUIs (app.py, media_gui.py)
# ---------------------------------------------------------------------------
# Werkzeug is pinned alongside Flask because Flask 3.x will not run against
# Werkzeug 2.x, and pip will happily resolve an old one from a stale cache.
flask>=3.0
werkzeug>=3.0

# ---------------------------------------------------------------------------
# File system support: exFAT, NTFS, HFS+, APFS, ext2/3/4 and the rest
# ---------------------------------------------------------------------------
# Without pytsk3, media_core falls back to fat32_core and can only walk FAT32
# volumes; carving and the SQLite blob extraction are unaffected.
#
# APFS additionally needs the underlying Sleuth Kit to be 4.7+ AND built with
# pool support. A pytsk3 built against an older libtsk installs fine and then
# silently finds no APFS volumes, so confirm before relying on it:
#   python3 media_core.py -i image.E01 --inspect -o /tmp
#
# Windows installs from a wheel. Linux and macOS compile against libtsk, so
# install the development package first or the build fails on a missing
# tsk/libtsk.h:
#   Debian/Ubuntu : sudo apt install libtsk-dev build-essential python3-dev
#   Fedora/RHEL   : sudo dnf install libtsk-devel gcc python3-devel
#   macOS         : brew install sleuthkit
pytsk3>=20230125

# ---------------------------------------------------------------------------
# EnCase E01 / Ex01 images
# ---------------------------------------------------------------------------
# Provides the 'pyewf' module. Without it .E01 files cannot be opened at all;
# --inspect reports the image as unreadable and names this package.
#
# Note the name mismatch: the distribution is libewf-python, the import is
# pyewf. If pip finds no wheel for your platform, take the matching build
# from https://github.com/libyal/libewf/releases rather than fighting the
# source build - that project ships prebuilt Windows binaries.
libewf-python>=20231119

# ---------------------------------------------------------------------------
# Virtual machine disk images (optional)
# ---------------------------------------------------------------------------
# Moved to requirements-vm.txt. pip aborts the entire install on the first
# package that fails to build, so a VMDK library nobody needed should not be
# able to take Flask and pytsk3 down with it:
#
#   pip install -r requirements-vm.txt
#
# Each is independent - install just the format you handle. Same naming
# pattern as above: distribution lib*-python, module py*.

# ---------------------------------------------------------------------------
# Windows wheels for the libyal packages
# ---------------------------------------------------------------------------
# The libyal projects publish per-Python-version wheels rather than a
# universal one, and on Windows-on-ARM pip would otherwise try a win_amd64
# wheel and fail. If you keep a wheel cache beside this file, pin it with
# markers on both the OS and the machine so ARM falls through to the sdist:
#
# whl_files/pytsk3-20230125-cp312-cp312-win_amd64.whl;        python_version == "3.12" and platform_system == "Windows" and platform_machine == "AMD64"
# whl_files/libewf_python-20231119-cp312-cp312-win_amd64.whl; python_version == "3.12" and platform_system == "Windows" and platform_machine == "AMD64"

# ---------------------------------------------------------------------------
# Deliberately absent
# ---------------------------------------------------------------------------
# lxml    - all XML is written as text with explicit UTF-16 encoding and CRLF
#           endings to match C4All byte for byte; a serialiser would get in
#           the way rather than help.
# sqlite3 - stdlib, and db_extract.py does not use it anyway: it parses the
#           file format directly so it can follow overflow chains and read
#           free space, neither of which the sqlite3 module exposes.

# ---------------------------------------------------------------------------
# Pictures: thumbnails, previews, screening, visual similarity, HEIC
# ---------------------------------------------------------------------------
# Pillow decodes images for the gallery's thumbnails, the reports' contact
# sheets, screening, and grouping look-alike pictures -- a re-saved JPEG, a
# resized copy, a screenshot of the same photo -- which hashes cannot find.
# Without it those features say they are unavailable rather than finding
# nothing. It decodes image data from the exhibit, so keep it current.
pillow>=10.0
# HEIC/HEIF, which iPhones write. Without it HEIC files are recovered and
# hashed but cannot be previewed, thumbnailed or compared.
pillow-heif>=0.16

# ---------------------------------------------------------------------------
# Video frames, face and content search, and faster MPEG-TS carving
# ---------------------------------------------------------------------------
# OpenCV pulls frames from videos a browser cannot play, and runs face and
# content search. Content search with the DINOv2 model needs OpenCV 5.0 or
# newer (MANUAL.md); everything else works on 4.8. NumPy is listed on its
# own because the carve uses it without OpenCV: MPEG-TS detection with it
# is one array pass, and without it the fallback walk was measured at 77%
# of a whole run.
opencv-python-headless>=4.8
numpy>=1.24

# ---------------------------------------------------------------------------
# Apple GPU textures (iOS app-switcher snapshots, textures inside apps)
# ---------------------------------------------------------------------------
# A .ktx or AAPL texture is recovered and hashed without these; they are what
# turn it into a thumbnail. texture2ddecoder decodes the ASTC/ETC/PVRTC block
# data, always in a separate process because it can crash on malformed input
# (fme/engines/textures.py). pyliblzfse unpacks the LZFSE compression most
# iOS textures carry first; PyPI has Windows wheels for Python 3.10-3.13 only,
# so on 3.14 or Windows on ARM it builds from source and needs a compiler.
# Without them a texture's tile says which package it needs.
texture2ddecoder>=1.0
pyliblzfse>=0.4.1

# ---------------------------------------------------------------------------
# Encrypted evidence
# ---------------------------------------------------------------------------
# The vendored readers decrypt AES/3DES-encrypted acquisitions, AFF sealed
# to a certificate, and BitLocker with pycryptodome (or pycryptodomex, which
# they also accept). Without it those images open as unreadable.
pycryptodome>=3.19
# LUKS2 volumes whose keyslots use Argon2 -- the default since cryptsetup
# 2.1 -- need argon2-cffi to derive the key. Without it they report
# "its Argon2 keyslot needs the argon2-cffi package"; PBKDF2 keyslots still
# open.
argon2-cffi>=23.1
# Checking Content Credentials (C2PA) signatures, and making the
# certificate for a shared hash store served over TLS. Without it C2PA
# claims are shown unverified and `fme hashserver --tls` cannot start.
cryptography>=42.0

# ---------------------------------------------------------------------------
# Archives and app caches
# ---------------------------------------------------------------------------
# Media inside 7z archives. Without it 7z archives are counted as skipped.
py7zr>=0.20
# Snapchat content bundles: files under com.snap.file_manager_*_SCContent_
# are often an LZC container of Zstandard frames. Without a decompressor
# they are counted and reported, not read. (Python 3.14 and later have one
# in the standard library, and it is used automatically when present.)
zstandard>=0.22

# ---------------------------------------------------------------------------
# Serving: the GUI and the shared hash store
# ---------------------------------------------------------------------------
# Flask's own server is single-threaded, so a gallery fetching many
# thumbnails at once stalls, and it prints a development-server warning.
# Waitress has neither limitation and is used automatically when present.
waitress>=2.1
# The shared hash store over TLS, threaded (with cryptography above).
cheroot>=10.0

# ---------------------------------------------------------------------------
# Reports and the window
# ---------------------------------------------------------------------------
# Writing PDF reports directly. Without it, open the HTML report in a
# browser and print it to PDF.
reportlab>=4.0
# A window of its own rather than a browser tab. Windows and macOS only:
# on Linux it needs GTK or Qt bindings installed separately, and with
# neither present the window fails to open, so it is left for Linux users
# to add on purpose (pip install pywebview, plus pygobject or qtpy).
pywebview>=5.0; platform_system == "Windows" or platform_system == "Darwin"
